cmake_minimum_required(VERSION 3.16)
project(ROBContinuousArm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROB_RUCKIG_SOURCE "" CACHE PATH "Offline checkout of Ruckig v0.9.2")
set(ROB_SANITIZE "" CACHE STRING "address,undefined or thread; empty disables")
if(ROB_SANITIZE)
  add_compile_options(-fsanitize=${ROB_SANITIZE} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${ROB_SANITIZE})
endif()
if(ROB_RUCKIG_SOURCE)
  file(READ "${ROB_RUCKIG_SOURCE}/CMakeLists.txt" RUCKIG_CMAKE)
  if(NOT RUCKIG_CMAKE MATCHES "project\\(ruckig VERSION 0\\.9\\.2 LANGUAGES")
    message(FATAL_ERROR "Use Ruckig 0.9.2, the installed Amber version")
  endif()
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_ONLINE_CLIENT OFF CACHE BOOL "" FORCE)
  set(BUILD_PYTHON_MODULE OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ROB_RUCKIG_SOURCE}" ruckig EXCLUDE_FROM_ALL)
else()
  find_package(ruckig 0.9.2 EXACT REQUIRED)
endif()
find_package(Threads REQUIRED)
add_library(rob_continuous_arm ContinuousArmController.cpp)
target_include_directories(rob_continuous_arm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rob_continuous_arm PUBLIC ruckig::ruckig Threads::Threads)
target_compile_options(rob_continuous_arm PRIVATE -Wall -Wextra -Werror)
add_library(rob_amber_wire_codec AmberWireCodec.cpp)
target_include_directories(rob_amber_wire_codec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(rob_amber_wire_codec PRIVATE -Wall -Wextra -Werror)
add_library(rob_receive_clock ReceiveClock.cpp)
target_include_directories(rob_receive_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(rob_receive_clock PRIVATE -Wall -Wextra -Werror)
add_library(rob_arm_driver_boundary ArmDriverBoundary.cpp)
target_include_directories(rob_arm_driver_boundary PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rob_arm_driver_boundary PUBLIC rob_amber_wire_codec rob_continuous_arm)
target_compile_options(rob_arm_driver_boundary PRIVATE -Wall -Wextra -Werror)
add_executable(continuous_arm_tests ../../Tests/ROBContinuousArmControllerTests.cpp)
target_link_libraries(continuous_arm_tests PRIVATE rob_continuous_arm)
target_compile_options(continuous_arm_tests PRIVATE -Wall -Wextra -Werror)
add_executable(legacy_handoff_reproduction ../../Tests/ROBAmberTrajectoryHandoffSimulation.cpp)
target_link_libraries(legacy_handoff_reproduction PRIVATE ruckig::ruckig)
add_executable(amber_wire_codec_tests ../../Tests/ROBAmberWireCodecTests.cpp)
target_link_libraries(amber_wire_codec_tests PRIVATE rob_amber_wire_codec rob_continuous_arm)
target_compile_options(amber_wire_codec_tests PRIVATE -Wall -Wextra -Werror)
add_executable(arm_driver_boundary_tests ../../Tests/ROBArmDriverBoundaryTests.cpp)
target_link_libraries(arm_driver_boundary_tests PRIVATE rob_arm_driver_boundary)
target_compile_options(arm_driver_boundary_tests PRIVATE -Wall -Wextra -Werror)
add_executable(receive_clock_tests ../../Tests/ROBReceiveClockTests.cpp)
target_link_libraries(receive_clock_tests PRIVATE rob_receive_clock)
target_compile_options(receive_clock_tests PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME continuous_arm COMMAND continuous_arm_tests)
add_test(NAME legacy_handoff COMMAND legacy_handoff_reproduction)
add_test(NAME amber_wire_codec COMMAND amber_wire_codec_tests)
add_test(NAME arm_driver_boundary COMMAND arm_driver_boundary_tests)
add_test(NAME receive_clock COMMAND receive_clock_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(rob_socketcan_transport SocketCanTransport.cpp)
  target_include_directories(rob_socketcan_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(rob_socketcan_transport PUBLIC rob_arm_driver_boundary rob_receive_clock)
  target_compile_options(rob_socketcan_transport PRIVATE -Wall -Wextra -Werror)
  add_executable(amber_linux_transport_tests ../../Tests/ROBAmberLinuxTransportTests.cpp)
  target_link_libraries(amber_linux_transport_tests PRIVATE rob_socketcan_transport)
  target_compile_options(amber_linux_transport_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME amber_linux_transport_no_io COMMAND amber_linux_transport_tests)
endif()
